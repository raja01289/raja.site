let cart = [];

function renderProducts(items) {
    const grid = document.getElementById('productGrid');
    grid.innerHTML = '';
    items.forEach(p => {
        grid.innerHTML += `
            <div class="product-card">
                <img src="${p.image}" alt="${p.title}">
                <div class="product-info">
                    <h3>${p.title}</h3>
                    <div class="product-price">Rs. ${p.price}</div>
                    <button class="btn-add" onclick="addToCart(${p.id})">Add to Cart</button>
                </div>
            </div>`;
    });
}

function addToCart(id) {
    const item = products.find(p => p.id === id);
    const found = cart.find(p => p.id === id);
    if (found) found.qty += 1;
    else cart.push({ ...item, qty: 1 });
    updateCartUI();
    toggleCart(true);
}

function updateCartUI() {
    const itemsDiv = document.getElementById('cartItems');
    const countSpan = document.getElementById('cartCount');
    const totalSpan = document.getElementById('cartTotal');
    itemsDiv.innerHTML = '';
    let total = 0, count = 0;
    
    if(cart.length === 0) {
        itemsDiv.innerHTML = '<p class="empty-msg">Your bag is empty.</p>';
    } else {
        cart.forEach((item, index) => {
            total += item.price * item.qty;
            count += item.qty;
            itemsDiv.innerHTML += `<div style="display:flex; justify-content:space-between; margin-bottom:10px;"><span>${item.title} (x${item.qty})</span><span>Rs. ${item.price * item.qty}</span></div>`;
        });
    }
    countSpan.innerText = count;
    totalSpan.innerText = `Rs. ${total}`;
}

function toggleCart(open = false) {
    const drawer = document.getElementById('cartDrawer');
    const overlay = document.getElementById('cartOverlay');
    if(open) {
        drawer.classList.add('active');
        overlay.style.display = 'block';
    } else {
        drawer.classList.toggle('active');
        overlay.style.display = drawer.classList.contains('active') ? 'block' : 'none';
    }
}

function searchProducts() {
    const q = document.getElementById('searchInput').value.toLowerCase();
    renderProducts(products.filter(p => p.title.toLowerCase().includes(q)));
}

function checkoutWhatsApp() {
    if(cart.length === 0) { alert('Cart is empty!'); return; }
    const phone = "923405049439"; 
    let msg = "Order from KSAH Sports PK:\n\n";
    let total = 0;
    cart.forEach(i => { msg += `- ${i.title} (x${i.qty}): Rs. ${i.price * i.qty}\n`; total += i.price * i.qty; });

}
